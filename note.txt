ATOM MyRegisterClass(HINSTANCE hInstance)
{
    WNDCLASSEXW wcex;

    wcex.cbSize = sizeof(WNDCLASSEX);

    wcex.style          = CS_HREDRAW | CS_VREDRAW;
    wcex.lpfnWndProc    = WndProc;
    wcex.cbClsExtra     = 0;
    wcex.cbWndExtra     = 0;
    wcex.hInstance      = hInstance;
    wcex.hIcon          = LoadIcon(hInstance, MAKEINTRESOURCE(IDI_RESOURCEPRACTICE));
    wcex.hCursor        = LoadCursor(nullptr, IDC_ARROW);
    wcex.hbrBackground  = (HBRUSH)(COLOR_WINDOW+1);
	
    wcex.lpszMenuName   = MAKEINTRESOURCEW(IDC_RESOURCEPRACTICE); // 알아낸 키 이름으로 바꾸면 생성한 메뉴 리소스로 변경
	
    wcex.lpszClassName  = szWindowClass;
    wcex.hIconSm        = LoadIcon(wcex.hInstance, MAKEINTRESOURCE(IDI_SMALL));

    return RegisterClassExW(&wcex);
}

 // 파일 탐색기 호출 함수
 void FileOpenDialog(HWND hWnd) // 메뉴 선택 시 호출될 함수 선언
{
    OPENFILENAME OFN;
    memset(&OFN, 0, sizeof(OPENFILENAME));
    TCHAR lpstrFile[256] = _T(""); // 파일 주소 텍스트 출력 크기
	TCHAR filter[] = _T("Every file(*.*)\0 *.*\0Text file\0*.txt;*.doc\0"); // 파일 종류 필터 설정, 정규표현식 참고

    OFN.lStructSize = sizeof(OPENFILENAME);
    OFN.hwndOwner = hWnd;
    OFN.lpstrFile = lpstrFile;
    OFN.nMaxFile = 100;
    OFN.lpstrInitialDir = _T("."); // 파일 위치 지정 .  ../  등
	OFN.lpstrFilter = filter;

    if(GetOpenFileName(&OFN))
    {
        MessageBox(hWnd, OFN.lpstrFile, _T("열기선택파일"), MB_OK);
    }
}

COLORREF g_SelectColor;
void OpenColorPicker(HWND hWnd)
{
    CHOOSECOLOR COLOR;
    static COLORREF tmp[16], color;
    HBRUSH hBrush, OldBrush;
    
    for (int i = 0; i < 16; i++)
        tmp[i] = RGB(rand() % 256, rand() % 256, rand() % 256);
    memset(&COLOR, 0, sizeof(CHOOSECOLOR));
    COLOR.lStructSize = sizeof(CHOOSECOLOR);
    COLOR.hwndOwner = hWnd;
    COLOR.lpCustColors = tmp;
    COLOR.Flags = CC_RGBINIT;
    if (ChooseColor(&COLOR) != 0)
    {
        g_SelectColor = COLOR.rgbResult;
        InvalidateRgn(hWnd, NULL, true);
    }
}
