// 그림 출력하기
HDC hdc = GetDC(hWnd); // 핸들 생성

ReleaseDC(hWnd, hdc); // 핸들 해제

InvalidateRgn(hWnd, NULL, true); // 화면 초기화 및 출력 호출

// 선 출력
MoveToEx() 선 시작점 지정
LineTo() 끝점 지정
ex)
MoveToEx(hdc, 10, 10, NULL);
LineTo(hdc, 100, 100);

// 원 출력
Ellipse()

// 사각형 출력
Rectangle()

// 다각형 출력
Polygon()
ex)
POINT pos[10] = { {10, 150}, {250, 30}, {500, 150}, {350, 300}, {150, 300} };
Polygon(hdc, pos, 5);

// 도형 그리기의 경우 내부도 채워지며 그려짐

// 선 관련 값
HPEN pen = CreatePen() // 펜 스타일, 굵기, 색상의 새로운 펜 생성
HPEN oldpen = (HPEN)SelectObject() // 펜 지정

SelectObject(hdc, oldpen); // 복구
DeleteObject(pen);

// 브러쉬 생성 및 설정(안에 면 채우기)
DWORD bcol = 0;
HBRUSH hbrush = CreateSolidBrush(RGB(128, 0, 0));
HBRUSH oldbrush = (HBRUSH)SelectObject(hdc, hbrush);

SelectObject(hdc, oldbrush); // 복구
DeleteObject(hbrush);

색상 코드 unsinged long 형태로 선언 / unsinged long col = RGB(255, 0 , 0);

// 사각형 그리기

// 선 그리기
